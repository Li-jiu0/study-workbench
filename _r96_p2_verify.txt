  /**
   * 安全截断：不切开代理对（surrogate pair），避免末尾出现孤立代理项渲染成乱码。
   * @param {String}  s 原串
   * @param {number}  n 期望长度上限
   * @returns {String}
   */
  function _safeCut(s, n) {
    if (s.length <= n) return s;
    var cut = s.slice(0, n);
    // 若末尾是高代理项（U+D800~U+DBFF），说明正好把一对字符切开了 -> 回退一位
    if (/[\uD800-\uDBFF]$/.test(cut)) cut = cut.slice(0, n - 1);
    return cut;
  }
